{{- $token := os.Getenv "GITHUB_TOKEN" -}} {{- $headers := dict "Accept" "application/json" -}} {{- $headers := cond (gt ($token | len) 1) (merge $headers (dict "Authorization" $token)) $headers -}} {{- $owner := .Get "owner" | default "DanySK" -}} {{- $repo := .Get "repo" | default "Template-Hugo-Reveal-Slides" -}} {{- $path := .Get "path" | default "LICENSE" -}} {{- $branch := .Get "branch" | default "master" -}} {{- $url := printf "https://api.github.com/repos/%s/%s/contents/%s?ref=%s" $owner $repo $path $branch -}} {{- $json := resources.GetRemote $url (dict "headers" $headers) | transform.Unmarshal -}} {{- $content := $json.content | base64Decode | htmlUnescape -}} {{- $content := split $content "\n" -}} {{- $start := .Get "from" | default "1" | int | add -1 }} {{- $end := .Get "to" | default (len $content) | int }} {{- $end := sub $end $start }} {{- $content := after $start (first (add $start $end) $content) -}} {{- $content := delimit $content "\n" -}} {{- $language := .Get "language" | default ( cond (or (strings.HasSuffix $path ".kt") (strings.HasSuffix $path ".kts")) "kotlin" ( cond (strings.HasSuffix $path ".yml") "yaml" ( cond (strings.HasSuffix $path ".md") "markdown" ( replaceRE "^.*\\.(\\w+)$" "$1" $path ) ) ) ) -}} {{- highlight ($content) ($language) "" -}}
